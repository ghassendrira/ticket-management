    @media (max-width:560px){.teams-page{padding:20px 16px}.checkbox-grid{grid-template-columns:1fr}.modal{border-radius:20px}.modal-header,.modal-body,.modal-footer{padding-left:18px;padding-right:18px}}
  `]
})
export class TeamManagementComponent implements OnInit {
  private readonly assignmentService = inject(AssignmentService);
  private readonly userService = inject(UserService);

  readonly CATEGORY_LIST = CATEGORY_LIST;
  readonly PRESET_SKILLS = PRESET_SKILLS;

  readonly teams = signal<TeamResponse[]>([]);
  readonly selectedTeamId = signal<string | null>(null);
  readonly selectedTeam = signal<TeamResponse | null>(null);
  readonly teamAgents = signal<AgentProfileResponse[]>([]);

  readonly users = signal<UserResponse[]>([]);
  readonly userMap = signal<Record<string, UserResponse>>({});
  readonly availableAgents = signal<UserResponse[]>([]);
  readonly pendingAgentId = signal<string | null>(null);

  readonly globalAlert = signal<{ type: 'success' | 'error' | 'info' | 'warning'; message: string } | null>(null);

  readonly loading = {
    teams: false,
    teamAgents: false,
    availableAgents: false,
    addAgent: false,
    removeAgent: {} as Record<string, boolean>,
    skills: {} as Record<string, boolean>,
    deleteSkill: {} as Record<string, boolean>
  };

  readonly teamModal = {
    open: false,
    team: null as TeamResponse | null,
    loading: false,
    form: { name: '', managedCategories: [] as string[] },
    alert: null as { type: 'success' | 'error' | 'info' | 'warning'; message: string } | null
  };

  readonly deleteTeamModal = {
    open: false,
    team: null as TeamResponse | null,
    loading: false,
    alert: null as { type: 'success' | 'error' | 'info' | 'warning'; message: string } | null
  };

  readonly skillsModal = {
    open: false,
    agent: null as AgentProfileResponse | null,
    loading: false,
    form: { skillName: 'JAVA', level: 1 } as AgentSkillRequest,
    alert: null as { type: 'success' | 'error' | 'info' | 'warning'; message: string } | null
  };

  ngOnInit() {
    this.loadAll();
  }

  async loadAll() {
    this.loading.teams = true;
    this.loading.availableAgents = true;
    try {
      const [teams, users] = await Promise.all([
        firstValueFrom(this.assignmentService.getTeams()),
        firstValueFrom(this.userService.getUsers(undefined, 'AGENT', true))
      ]);
      this.teams.set(teams);
      this.users.set(users);
      const map: Record<string, UserResponse> = {};
      users.forEach(u => (map[u.id] = u));
      this.userMap.set(map);

      if (this.selectedTeamId()) {
        this.selectTeam(this.selectedTeamId()!);
      }
      this.refreshAvailableAgents();
    } catch (e: any) {
      this.setGlobalAlert('error', 'Impossible de charger les donnÃ©es. VÃ©rifiez que assignment-service est dÃ©marrÃ©.');
      console.error(e);
    } finally {
      this.loading.teams = false;
      this.loading.availableAgents = false;
    }
  }

  refreshAvailableAgents() {
    const agentsWithProfile = new Set<string>();
    const assignedUserIds = new Set<string>();

    for (const team of this.teams()) {
      // not fully accurate without agents loaded; we'll refresh after selecting.
    }

    // Instead, fetch agent profiles to know who is already activated/assigned
    firstValueFrom(this.assignmentService.getAgents())
      .then(profiles => {
        profiles.forEach(p => {
          agentsWithProfile.add(p.userId);
          if (p.teamId) assignedUserIds.add(p.userId);
        });

        const activeUsers = this.users();
        const available = activeUsers.filter(u => {
          const hasProfile = agentsWithProfile.has(u.id);
          if (!hasProfile) return true; // can be activated
          return !assignedUserIds.has(u.id);
        });

        this.availableAgents.set(available);
      })
      .catch(err => {
        // If service is down, fall back to listing users without filtering
        this.availableAgents.set(this.users());
        console.warn('Could not load agent profiles for availability filter', err);
      });
  }

  userById(userId: string): UserResponse | undefined {
    return this.userMap()[userId];
  }

  async selectTeam(id: string) {
    this.selectedTeamId.set(id);
    this.selectedTeam.set(this.teams().find(t => t.id === id) || null);
    this.loading.teamAgents = true;
    this.teamAgents.set([]);
    try {
      const agents = await firstValueFrom(this.assignmentService.getTeamAgents(id));
      this.teamAgents.set(agents);
    } catch (e: any) {
      this.setGlobalAlert('error', "Impossible de charger les agents de l'Ã©quipe.");
      console.error(e);
    } finally {
      this.loading.teamAgents = false;
    }
  }

  openTeamModal(team?: TeamResponse, event?: MouseEvent) {
    if (event) event.stopPropagation();
    this.teamModal.alert = null;
    this.teamModal.loading = false;
    this.teamModal.team = team || null;
    this.teamModal.form = {
      name: team?.name || '',
      managedCategories: team?.managedCategories ? [...(team.managedCategories as unknown as string[])] : []
    };
    this.teamModal.open = true;
  }

  closeTeamModal() {
    this.teamModal.open = false;
    this.teamModal.team = null;
    this.teamModal.alert = null;
  }

  toggleCategory(cat: string) {
    const cur = this.teamModal.form.managedCategories;
    if (cur.includes(cat)) {
      this.teamModal.form.managedCategories = cur.filter(c => c !== cat);
    } else {
      this.teamModal.form.managedCategories = [...cur, cat];
    }
  }

  canSaveTeam(): boolean {
    const f = this.teamModal.form;
    return !!(f.name && f.name.trim().length > 0 && f.managedCategories.length > 0);
  }

  async saveTeam() {
    if (!this.canSaveTeam()) return;
    this.teamModal.loading = true;
    this.teamModal.alert = null;
    const request: TeamRequest = {
      name: this.teamModal.form.name.trim(),
      managedCategories: this.teamModal.form.managedCategories
    };
    try {
      if (this.teamModal.team) {
        await firstValueFrom(this.assignmentService.updateTeam(this.teamModal.team.id, request));
        this.setGlobalAlert('success', 'Ã‰quipe mise Ã  jour avec succÃ¨s.');
      } else {
        const created = await firstValueFrom(this.assignmentService.createTeam(request));
        this.setGlobalAlert('success', 'Ã‰quipe crÃ©Ã©e avec succÃ¨s.');
        this.selectedTeamId.set(created.id);
      }
      this.closeTeamModal();
      await this.loadAll();
    } catch (e: any) {
      const msg = e?.error?.message || "Erreur lors de l'enregistrement.";
      this.teamModal.alert = { type: 'error', message: msg };
    } finally {
      this.teamModal.loading = false;
    }
  }

  openDeleteTeamModal(team: TeamResponse, event?: MouseEvent) {
    if (event) event.stopPropagation();
    this.deleteTeamModal.team = team;
    this.deleteTeamModal.open = true;
    this.deleteTeamModal.loading = false;
    this.deleteTeamModal.alert = null;
  }

  closeDeleteTeamModal() {
    this.deleteTeamModal.open = false;
    this.deleteTeamModal.team = null;
    this.deleteTeamModal.alert = null;
  }

  async confirmDeleteTeam() {
    if (!this.deleteTeamModal.team) return;
    this.deleteTeamModal.loading = true;
    this.deleteTeamModal.alert = null;
    try {
      await firstValueFrom(this.assignmentService.deleteTeam(this.deleteTeamModal.team.id));
      if (this.selectedTeamId() === this.deleteTeamModal.team.id) {
        this.selectedTeamId.set(null);
        this.selectedTeam.set(null);
        this.teamAgents.set([]);
      }
      this.setGlobalAlert('success', 'Ã‰quipe supprimÃ©e avec succÃ¨s.');
      this.closeDeleteTeamModal();
      await this.loadAll();
    } catch (e: any) {
      const msg = e?.error?.message || 'Impossible de supprimer cette Ã©quipe.';
      this.deleteTeamModal.alert = { type: 'error', message: msg };
    } finally {
      this.deleteTeamModal.loading = false;
    }
  }

  async addAgentToTeam() {
    const userId = this.pendingAgentId();
    const teamId = this.selectedTeamId();
    if (!userId || !teamId) return;

    this.loading.addAgent = true;
    try {
      // ensure agent profile exists
      let profile: AgentProfileResponse;
      const existing = (await firstValueFrom(this.assignmentService.getAgents())).find(a => a.userId === userId);
      if (!existing) {
        profile = await firstValueFrom(
          this.assignmentService.createAgentProfile({ userId, maxConcurrentTickets: 5 })
        );
      } else {
        profile = existing;
      }

      // assign to team
      await firstValueFrom(
        this.assignmentService.updateAgentTeam(profile.id, { teamId })
      );

      this.pendingAgentId.set(null);
      this.setGlobalAlert('success', 'Agent ajoutÃ© Ã  l\'Ã©quipe avec succÃ¨s.');
      await this.loadAll();
    } catch (e: any) {
      const msg = e?.error?.message || "Erreur lors de l'ajout de l'agent.";
      this.setGlobalAlert('error', msg);
    } finally {
      this.loading.addAgent = false;
    }
  }

  async removeAgentFromTeam(agent: AgentProfileResponse) {
    this.loading.removeAgent[agent.id] = true;
    try {
      await firstValueFrom(this.assignmentService.updateAgentTeam(agent.id, { teamId: null }));
      this.setGlobalAlert('success', 'Agent retirÃ© de l\'Ã©quipe.');
      await this.loadAll();
    } catch (e: any) {
      const msg = e?.error?.message || 'Erreur.';
      this.setGlobalAlert('error', msg);
    } finally {
      this.loading.removeAgent[agent.id] = false;
    }
  }

  openSkillsModal(agent: AgentProfileResponse) {
    this.loading.skills[agent.id] = false;
    this.skillsModal.agent = JSON.parse(JSON.stringify(agent));
    this.skillsModal.open = true;
    this.skillsModal.loading = false;
    this.skillsModal.alert = null;
    this.skillsModal.form = { skillName: 'JAVA', level: 3 };
  }

  closeSkillsModal() {
    this.skillsModal.open = false;
    this.skillsModal.agent = null;
    this.skillsModal.alert = null;
    void this.loadAll();
  }

  async confirmAddSkill() {
    if (!this.skillsModal.agent) return;
    this.skillsModal.loading = true;
    this.skillsModal.alert = null;
    try {
      const updated = await firstValueFrom(
        this.assignmentService.addSkill(this.skillsModal.agent.id, {
          skillName: this.skillsModal.form.skillName.toUpperCase(),
          level: Number(this.skillsModal.form.level)
        })
      );
      this.skillsModal.agent = updated;
      this.skillsModal.alert = { type: 'success', message: 'CompÃ©tence ajoutÃ©e / mise Ã  jour.' };
    } catch (e: any) {
      const msg = e?.error?.message || 'Erreur.';
      this.skillsModal.alert = { type: 'error', message: msg };
    } finally {
      this.skillsModal.loading = false;
    }
  }

  async confirmDeleteSkill(skill: AgentSkillResponse) {
    if (!this.skillsModal.agent) return;
    this.loading.deleteSkill[skill.id] = true;
    try {
      await firstValueFrom(
        this.assignmentService.deleteSkill(this.skillsModal.agent.id, skill.id)
      );
      const refreshed = await firstValueFrom(this.assignmentService.getAgent(this.skillsModal.agent.id));
      this.skillsModal.agent = refreshed;
    } catch (e: any) {
      const msg = e?.error?.message || 'Erreur.';
      this.skillsModal.alert = { type: 'error', message: msg };
    } finally {
      this.loading.deleteSkill[skill.id] = false;
    }
  }

  getBadgeStyle(cat: string): string {
    const map: CategoryBadgeMap = {
      ACCOUNT_ACCESS: 'ACCOUNT_ACCESS',
      BILLING: 'BILLING',
      TECHNICAL: 'TECHNICAL',
      ORDER: 'ORDER',
      DELIVERY: 'DELIVERY',
      SECURITY: 'SECURITY',
      INFORMATION: 'INFORMATION',
      OTHER: 'OTHER'
    };
    return map[cat] || 'OTHER';
  }

  private setGlobalAlert(type: 'success' | 'error' | 'info' | 'warning', message: string) {
    this.globalAlert.set({ type, message });
    setTimeout(() => {
      if (this.globalAlert()?.message === message) this.globalAlert.set(null);
    }, 6000);
  }
}
